Домашнее задание. Фигуры:  
Условие задачи. Составьте программу, которая для заданной пользователем фигуры, например (прямоугольника, треугольника, трапеции, квадрата и круга) предлагает меню выбора одной из операций:
1) рассчитать площадь;
2) вывести определение фигуры;
3) нарисовать фигуру.                                                                                               
Для каждой из операций и фигур реализуйте одну собственную функцию вычисления площади с необходимым числом параметров, например :
float area_round(float a); // для круга
float area_rectangle(float a, float b); // для прямоугольника
float area_triangle(float a, float b, float c); // для треугольника
и рисования  фигуры символом '*' (размеры сторон фигуры пользователь задает в main):
void draw_rectangle(int a, int b);
или рисования  фигуры (вид символа пользователь задает в main):
void draw_round(char a);
Для вывода параметров фигуры: 
int get_radius(int a); // возвращает радиус круга
Для сравнения двух фигур по площади
int compare_round(int a, int b)// возвращает разницу между двумя кругами с радиусами a и b.

float area_round(float r) 
{
    return (float)M_PI * pow(r, 2.0);
}

float area_rectangle(float a, float b) 
{
    return a * b;
}

float area_triangle(float a, float b, float c) 
{
    if (a + b <= c || a + c <= b || b + c <= a) 
    {
        printf("Ошибка: треугольник с такими сторонами не существует!\n");
        return -1;
    }
    float p = (a + b + c) / 2.0f;
    return sqrtf(p * (p - a) * (p - b) * (p - c));
}

float area_trapezoid(float a, float b, float h) 
{
    return ((a + b) / 2.0f) * h;
}

float area_square(float a) 
{
    return pow(a, 2.0);
}

void print_definition(int figure_type) 
{
    printf("Определение: ");
    if (figure_type == 1) puts("Прямоугольник — это четырехугольник, у которого все углы прямые, а стороны попарно равны");
    else if (figure_type == 2) puts("Треугольник — это геометрическая фигура, образованная тремя отрезками, соединяющими три точки, не лежащие на одной прямой.");
    else if (figure_type == 3) puts("Трапеция — это четырехугольник, у которого две стороны параллельны, а две другие не параллельны.");
    else if (figure_type == 4) puts("Квадрат — это правильный четырехугольник, у которого все стороны и углы равны.");
    else if (figure_type == 5) puts("Круг — это часть плоскости, ограниченная окружностью.");
}

void draw_rectangle(int a, int b) 
{
    printf("\n");
    for (int i = 0; i < b; i++) 
    {
        for (int j = 0; j < a; j++) 
        {
            printf("* ");
        }
        printf("\n");
    }
}

void draw_square(int a) 
{
    draw_rectangle(a, a);
}

void draw_triangle(int h) 
{
    printf("\n");
    for (int i = 1; i <= h; i++) 
    {
        for (int j = 0; j < i; j++) 
        {
            printf("* ");
        }
        printf("\n");
    }
}

void draw_trapezoid(int a, int b, int h) 
{
    printf("\n");
    for (int i = 0; i < h; i++) 
    {
        int c_width = a + (i * (b - a)) / (h > 1 ? h - 1 : 1);
        for (int j = 0; j < c_width; j++) 
        {
            printf("* ");
        }
        printf("\n");
    }
}

void draw_round(char symbol)
{
    int r = 4;
    printf("\n");
    for (int i = -r; i <= r; i++) 
    {
        for (int j = -r; j <= r; j++) 
        {
            if (i*i + j*j <= r*r) printf("%c ", symbol);
            else printf("  ");
        }
        printf("\n");
    }
}

int get_radius(int r) 
{
    return r;
}

int compare_round(int r1, int r2) 
{
    float s1 = area_round((float)r1);
    float s2 = area_round((float)r2);
    return (int)fabsf(s1 - s2);
}
