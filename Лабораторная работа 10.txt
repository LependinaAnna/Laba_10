double fa(double X) 
{
    double y;
    if (X <= 3.0) 
    {
        y = pow(X, 2.0) - 3.0 * X + 9.0;
    } 
    else 
    {
        y = 1.0 / (pow(X, 3.0) + 3.0);
    }
    return y;
}

double fb(double X) 
{
    double y;
    y = X * exp(sin(pow(X, 2.0)));
    return y;
}

double sin_n(double x, int N) 
{
    double sum, terminal; 
    sum = 0.0;
    terminal = x;
    for (int k = 1; k <= N; k++) 
    {
        sum += terminal;
        terminal = terminal * (-x * x) / ((2 * k) * (2 * k + 1));
    }
    return sum;
}

double sin_eps(double x, double eps) 
{
    double sum, terminal;
    sum = 0.0;
    terminal = x;
    int k = 1;
    while (fabs(terminal) > eps) 
    {
        sum += terminal;
        terminal = terminal * (-x * x) / ((2 * k) * (2 * k + 1));
        k++;
    }
    return sum;
}

double arcsin_n(double x_6, int N_6) 
{
    double sum_6, terminal_6; 
    sum_6 = x_6;
    terminal_6 = x_6;
    for (int n = 1; n < N_6; n++) 
    {
        terminal_6 = terminal_6 * (x_6 * x_6 * (2 * n - 1) * (2 * n - 1)) / ((2 * n) * (2 * n + 1));
        sum_6 += terminal_6;
    }
    return sum_6;
}

double arcsin_eps(double x_6, double eps_6) 
{
    double sum_6, terminal_6;
    sum_6 = x_6;
    terminal_6 = x_6;
    int n = 1;
    while (fabs(terminal_6) > eps_6) 
    {
        terminal_6 = terminal_6 * (x_6 * x_6 * (2 * n - 1) * (2 * n - 1)) / ((2 * n) * (2 * n + 1));
        sum_6 += terminal_6;
        n++;
    }
    return sum_6;
}
